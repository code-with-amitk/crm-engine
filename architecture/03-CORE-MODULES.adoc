[source, text]
----

  ↓

==== Error Mapping

The module maps database errors to user-friendly messages:

[source, go]
----
func HandleDBError(err error, logger) (int, string, string) {
    // SQL unique constraint violation
    if strings.Contains(err.Error(), "duplicate key") {
        return http.StatusConflict, "duplicate_entry", 
               "Record already exists"
    }
    
    // Foreign key violation
    if strings.Contains(err.Error(), "foreign key") {
        return http.StatusBadRequest, "invalid_reference",
               "Referenced record does not exist"
    }
    
    // Connection error
    if strings.Contains(err.Error(), "connection refused") {
        return http.StatusServiceUnavailable, "db_unavailable",
               "Database is not available"
    }
    
    // Default
    return http.StatusInternalServerError, "internal_error",
           "An error occurred"
}
----

==== Transaction Management

[source, go]
----
func ExecuteInTransaction(db *sql.DB, fn func(*sql.Tx) error) error {
    tx, err := db.BeginTx(ctx, nil)
    if err != nil {
        return err
    }
    defer tx.Rollback()
    
    if err := fn(tx); err != nil {
        return err
    }
    
    return tx.Commit().Error
}
----

=== Parameterized Queries (SQL Injection Prevention)

[source, go]
----
// ✅ SAFE: Uses parameters
query := "SELECT * FROM employee WHERE emp_email = $1"
row := db.QueryRow(query, userEmail)

// ❌ UNSAFE: String concatenation
query := fmt.Sprintf("SELECT * FROM employee WHERE emp_email = '%s'", userEmail)

## 3. JWT Module (`jwt_mod/`)

=== Purpose
Handles JWT token generation, validation, and claims management.

=== Key Functions

==== Token Generation

[source, go]
----
func GenerateJWTToken(claims *Claims, secret string) (string, error) {
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString([]byte(secret))
}
----

==== Token Validation

[source, go]
----
func ValidateJWT(logger, tokenString string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenString, &Claims{},
        func(token *jwt.Token) (interface{}, error) {
            return []byte(os.Getenv("JWT_SECRET")), nil
        })
    
    if err != nil {
        return nil, err
    }
    
    claims, ok := token.Claims.(*Claims)
    if !ok || !token.Valid {
        return nil, ErrInvalidToken
    }
    
    return claims, nil
}
----

==== Claims Structure

[source, go]
----
type Claims struct {
    EmployeeID int64
    TenantID int64
    RegisteredClaims jwt.RegisteredClaims
}

// Fields set during login:
// - EmployeeID: from employee table
// - TenantID: from employee table
// - ExpiresAt: NOW() + JWT_ACCESS_EXPIRY_HOURS
// - IssuedAt: NOW()
// - Subject: "access_token"
----

==== Context Management

[source, go]
----
func ContextWithClaims(ctx context.Context, claims *Claims) context.Context {
    return context.WithValue(ctx, "claims", claims)
}

func ClaimsFromContext(ctx context.Context) *Claims {
    claims := ctx.Value("claims")
    if claims == nil {
        return nil
    }
    return claims.(*Claims)
}
----

=== Token Lifecycle

[source, text]
----
1. LOGIN
   ├─> POST /login (email + password)
   ├─> Verify email exists
   ├─> Hash provided password
   ├─> Compare with stored hash
   └─> If match:
       ├─> Generate Access Token (24h expiry)
       ├─> Generate Refresh Token (90d expiry)
       └─> Return both tokens

2. ACCESS TOKEN USAGE
   ├─> Include in header: Authorization: Bearer {token}
   ├─> API validates signature
   ├─> Check expiration (24h)
   └─> Extract claims for user context

3. TOKEN REFRESH
   ├─> Access token expired (or about to expire)
   ├─> POST /token/refresh (with refresh token)
   ├─> Validate refresh token in database
   ├─> Check if revoked
   ├─> If valid:
   │   ├─> Generate new Access Token
   │   ├─> Optionally rotate Refresh Token
   │   └─> Return new Access Token
   └─> If invalid: Return 401

4. LOGOUT
   ├─> POST /logout (optional)
   ├─> Revoke refresh token in database
   └─> Discard access token on client
----

## 5. Rate Limiting Module (`ratelimit/`)

=== Purpose
Implements fixed-window rate limiting to prevent abuse.

=== Algorithm

**Fixed-Window Counter**:
* Divide time into fixed windows (1 minute)
* Count requests per user/IP in current window
* Reset at start of new window

[source, text]
----
Time
├─ Window 1: 00:00-00:59 (5 requests allowed, 3 used)
├─ Window 2: 01:00-01:59 (5 requests allowed, 4 used)
├─ Window 3: 02:00-02:59 (5 requests allowed, 5 used) ← LIMIT REACHED
│   └─ 6th request → REJECTED (429 Too Many Requests)
└─ Window 4: 03:00-03:59 (counter resets)
----

=== Implementation

[source, go]
----
type Limiter struct {
    mu      sync.Mutex
    buckets map[string]int  // IP/UserID → count
    limits  map[string]int  // IP/UserID → limit
}

func (l *Limiter) Allow(key string) bool {
    l.mu.Lock()
    defer l.mu.Unlock()
    
    if l.buckets[key] < l.limits[key] {
        l.buckets[key]++
        return true
    }
    return false
}
----

=== Rate Limits

[cols="30%,20%,50%"]
|===
| Endpoint | Limit | Purpose

| /login | 5(configurable) per minute (per IP) | Brute-force protection
| /token/refresh | 5(configurable) per minute (per IP) | Token refresh throttling
| All other authenticated | 50 per minute (per user) | General API protection
|===

=== Cleanup

[source, go]
----
func (l *Limiter) StartCleanup(interval time.Duration, logger) {
    go func() {
        ticker := time.NewTicker(interval)
        defer ticker.Stop()
        
        for range ticker.C {
            l.mu.Lock()
            // Clear old buckets
            for key := range l.buckets {
                l.buckets[key] = 0  // Reset counter
            }
            l.mu.Unlock()
        }
    }()
}
----

## 6. Logging Module (`logging/`)

=== Purpose
Configures structured JSON logging using Zap.

=== Configuration

[source, go]
----
type LogConfig struct {
    Level      string  // debug, info, warn, error
    OutputPath string  // "stdout" or filename
    ServiceID  int16   // 1=app, 2=notifier, 3=parser
}

func InitLogger(id int16) (*zap.SugaredLogger, error) {
    config := zap.NewProductionConfig()
    config.Level = zap.NewAtomicLevelAt(zapcore.InfoLevel)
    
    // Load from environment
    if level, ok := os.LookupEnv("APP_LOG_LEVEL"); ok {
        config.Level = parseLevel(level)
    }
    if output, ok := os.LookupEnv("APP_LOG_TO"); ok {
        config.OutputPaths = []string{output}
    }
    
    logger, _ := config.Build()
    return logger.Sugar(), nil
}

=== Notification Lifecycle

[source, text]
----
1. App registers push token
   └─> POST /register_push_token
   ├─> Mobile Device sends token: "ExponentPushToken[...]"
   └─> crm-engine Store in pushtokens table

2. Notifier daemon discovers due follow-up
   └─> Query WHERE followuptimestamp <= NOW()
   ├─> Retrieve employee's push tokens
   └─> Iterate each token

3. Send notification
   └─> Call SendExpoNotification()
   ├─> POST to Expo servers
   └─> Mark as notified in database

4. Device receives notification
   └─> Expo SDK shows alert
   └─> User taps to open app
   └─> App syncs data
